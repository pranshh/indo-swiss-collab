library(dplyr)
library(ggplot2)
library(VennDiagram)
library(RColorBrewer)
library(gridExtra)
library(scales)
library(ggalluvial)

# Calculate coverage statistics
coverage_stats <- dois %>%
  summarise(
    total_dois = n(),
    scopus_coverage = sum(Scopus, na.rm = TRUE),
    wos_coverage = sum(WoS, na.rm = TRUE),
    openalex_coverage = sum(openAlex, na.rm = TRUE),
    scopus_pct = round(100 * scopus_coverage / total_dois, 1),
    wos_pct = round(100 * wos_coverage / total_dois, 1),
    openalex_pct = round(100 * openalex_coverage / total_dois, 1)
  )

# Create overlap analysis
overlap_analysis <- dois %>%
  mutate(
    coverage_pattern = case_when(
      Scopus & WoS & openAlex ~ "All three",
      Scopus & WoS & !openAlex ~ "Scopus + WoS only",
      Scopus & !WoS & openAlex ~ "Scopus + OpenAlex only",
      !Scopus & WoS & openAlex ~ "WoS + OpenAlex only",
      Scopus & !WoS & !openAlex ~ "Scopus only",
      !Scopus & WoS & !openAlex ~ "WoS only",
      !Scopus & !WoS & openAlex ~ "OpenAlex only",
      TRUE ~ "None"
    )
  ) %>%
  count(coverage_pattern) %>%
  mutate(
    percentage = round(100 * n / sum(n), 1),
    coverage_pattern = factor(coverage_pattern, levels = c(
      "All three", "Scopus + WoS only", "Scopus + OpenAlex only", 
      "WoS + OpenAlex only", "Scopus only", "WoS only", "OpenAlex only", "None"
    ))
  )

# 1. Overall coverage bar chart
p1 <- coverage_stats %>%
  select(ends_with("_coverage")) %>%
  pivot_longer(everything(), names_to = "database", values_to = "count") %>%
  mutate(
    database = case_when(
      database == "scopus_coverage" ~ "Scopus",
      database == "wos_coverage" ~ "Web of Science",
      database == "openalex_coverage" ~ "OpenAlex"
    ),
    percentage = round(100 * count / coverage_stats$total_dois, 1)
  ) %>%
  ggplot(aes(x = reorder(database, count), y = count, fill = database)) +
  geom_col(width = 0.7, alpha = 0.8) +
  geom_text(aes(label = paste0(comma(count), "\n(", percentage, "%)")), 
            hjust = -0.1, size = 3.5, fontface = "bold") +
  scale_fill_manual(values = c("Scopus" = "#E31A1C", "Web of Science" = "#1F78B4", "OpenAlex" = "#33A02C")) +
  scale_y_continuous(labels = comma_format(), expand = expansion(mult = c(0, 0.15))) +
  coord_flip() +
  labs(
    title = "Database Coverage Comparison",
    subtitle = paste("Total DOIs analyzed:", comma(coverage_stats$total_dois)),
    x = "Database",
    y = "Number of DOIs Covered"
  ) +
  theme_minimal() +
  theme(
    legend.position = "none",
    plot.title = element_text(size = 14, face = "bold"),
    plot.subtitle = element_text(size = 12),
    axis.text = element_text(size = 11),
    axis.title = element_text(size = 12)
  )

# 2. Overlap patterns
p2 <- overlap_analysis %>%
  filter(coverage_pattern != "None") %>%
  ggplot(aes(x = reorder(coverage_pattern, n), y = n, fill = coverage_pattern)) +
  geom_col(width = 0.7, alpha = 0.8) +
  geom_text(aes(label = paste0(comma(n), "\n(", percentage, "%)")), 
            hjust = -0.1, size = 3.5, fontface = "bold") +
  scale_fill_brewer(type = "qual", palette = "Set2") +
  scale_y_continuous(labels = comma_format(), expand = expansion(mult = c(0, 0.15))) +
  coord_flip() +
  labs(
    title = "Database Overlap Patterns",
    subtitle = "Distribution of DOIs across different database combinations",
    x = "Coverage Pattern",
    y = "Number of DOIs"
  ) +
  theme_minimal() +
  theme(
    legend.position = "none",
    plot.title = element_text(size = 14, face = "bold"),
    plot.subtitle = element_text(size = 12),
    axis.text = element_text(size = 11),
    axis.title = element_text(size = 12)
  )

# 3. Venn diagram data preparation
venn_data <- dois %>%
  summarise(
    scopus_only = sum(Scopus & !WoS & !openAlex),
    wos_only = sum(!Scopus & WoS & !openAlex),
    openalex_only = sum(!Scopus & !WoS & openAlex),
    scopus_wos = sum(Scopus & WoS & !openAlex),
    scopus_openalex = sum(Scopus & !WoS & openAlex),
    wos_openalex = sum(!Scopus & WoS & openAlex),
    all_three = sum(Scopus & WoS & openAlex)
  )

# Create Venn diagram
venn_plot <- draw.triple.venn(
  area1 = coverage_stats$scopus_coverage,
  area2 = coverage_stats$wos_coverage,
  area3 = coverage_stats$openalex_coverage,
  n12 = venn_data$scopus_wos + venn_data$all_three,
  n23 = venn_data$wos_openalex + venn_data$all_three,
  n13 = venn_data$scopus_openalex + venn_data$all_three,
  n123 = venn_data$all_three,
  category = c("Scopus", "Web of Science", "OpenAlex"),
  fill = c("#E31A1C", "#1F78B4", "#33A02C"),
  alpha = 0.6,
  cex = 1.5,
  cat.cex = 1.2,
  cat.fontface = "bold",
  margin = 0.1
)


# Combine plots
grid.arrange(p1, p2, ncol = 2)
grid.draw(venn_plot)
